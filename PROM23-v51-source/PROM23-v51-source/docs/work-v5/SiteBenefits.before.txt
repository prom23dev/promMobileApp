  return <Pressable accessibilityRole="button" accessibilityLabel="Доставка, самовывоз и пункты выдачи" onPress={() => router.push('/information/delivery')} style={[layout.card, { padding: 16 }]}>
    <View style={[layout.row, { gap: 10 }]}><Truck size={22} color={colors.brand} /><View style={{ flex: 1 }}><Text style={type.subheading}>Получение · {state.city}</Text><Text style={type.caption}>Самовывоз, СДЭК, Яндекс Маркет и перевозчики</Text></View><ChevronRight size={18} color={colors.brand} /></View>
    {branch && <View style={[layout.row, { gap: 8 }]}><MapPin size={18} color={colors.brand} /><Text style={[type.caption, { flex: 1 }]}>{branch.address}</Text></View>}
    <Text style={type.caption}>Стоимость и срок зависят от склада, габаритов и маршрута. Доступность ПВЗ для вашего груза уточняется при оформлении.</Text>
